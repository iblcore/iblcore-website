{{- /* The resource pages one workflow step refers to. Takes a step map and returns a slice of pages. A path that resolves to nothing fails the build: a workflow with a missing step is worse than no workflow. */ -}} {{- $step := . -}} {{- $pages := slice -}} {{- range $path := partial "resources/step-paths.html" $step -}} {{- $page := site.GetPage $path -}} {{- if not $page -}} {{- errorf "workflow step %q refers to %q, which is not a resource page" $step.title $path -}} {{- end -}} {{- $pages = $pages | append $page -}} {{- end -}} {{- return $pages -}}
